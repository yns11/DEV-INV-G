# =============================================================================
# Databricks Apps runtime dependencies — Campagnes Inventaire
# -----------------------------------------------------------------------------
# The Databricks Apps runtime (Python 3.11 / Ubuntu 22.04) pre-installs a small
# set of packages, but NOT psycopg / databricks-sql-connector. Everything the
# app imports at runtime must be pinned here: there is no root access in the
# container, so only pure-Python or manylinux wheels from PyPI can be used.
#
# Pins use `~=` (compatible release) so that patch-level security fixes are
# picked up while the API surface stays stable.
# =============================================================================

# --- Web layer ---------------------------------------------------------------
fastapi~=0.115.0
uvicorn[standard]~=0.32.0
python-multipart~=0.0.12        # multipart/form-data uploads (scans, Excel files)
pydantic~=2.9.0
pydantic-settings~=2.6.0

# --- Databricks platform -----------------------------------------------------
databricks-sdk>=0.81,<1.0       # WorkspaceClient + w.postgres (Lakebase Autoscaling)
openai~=1.54                    # required by WorkspaceClient().serving_endpoints
                                # .get_open_ai_client(): the SDK imports `openai`
                                # and raises ImportError without it. Pulls httpx.
#
# NOT declared here: databricks-sql-connector. Nothing in the app imports it —
# cross-campaign Delta reads run in the Lakeflow job (Spark, jobs/), not in the
# app process. It also caps numpy below 2.0 up to and including 3.7.x, which is
# incompatible with the pins below and makes the whole install unresolvable. If
# a warehouse-backed read is ever added to the app, require version >= 4.

# --- Lakebase (PostgreSQL OLTP) ---------------------------------------------
psycopg[binary,pool]~=3.2.0     # psycopg 3: native connection pool + pipeline mode

# --- Data / analytics --------------------------------------------------------
numpy~=2.1.0
pandas~=2.2.0
scikit-learn~=1.5.0
scipy~=1.14.0

# --- File ingestion / reporting ---------------------------------------------
openpyxl~=3.1.5                 # .xlsx read (ERP exports) and write (downloads)
XlsxWriter~=3.2.0               # fast streaming .xlsx writer for large exports
reportlab~=4.2.0                # printable counting sheets (PDF)
pypdf~=5.1.0                    # split / inspect uploaded scanned sheets

# --- Misc --------------------------------------------------------------------
tenacity~=9.0.0                 # retry policy for serving-endpoint calls
